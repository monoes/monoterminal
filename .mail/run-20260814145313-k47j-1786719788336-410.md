    pub async fn create_session(
        &self,
        shell: String,
        working_dir: PathBuf,
        rows: u16,
        cols: u16,
    ) -> Result<SessionId, SessionError> {
        let id = Uuid::new_v4();

        let pty = PtyHandle::spawn(&shell, &working_dir, rows, cols).await?;
        
        let session = Session {
            id,
            pty,
            clients: Arc::new(RwLock::new(Vec::new())),
            scrollback: RingBuffer::new(10_000),
            // ...
        };
        
        // Spawn output fan-out task
        let session = Arc::new(RwLock::new(session));
        tokio::spawn(pty_output_loop(session.clone()));
        
        Ok(id)
    }
}

// Output fan-out loop (reads from your PtyHandle, broadcasts to clients)
async fn pty_output_loop(session: Arc<RwLock<Session>>) {
    loop {
        let chunk = {
            let mut s = session.write().await;
            s.pty.read().await  // ← Calls your read()
        };
        
        match chunk {
            Some(data) => {
                let shared = Arc::new(Bytes::from(data));
                let s = session.read().await;
                
                // Fan out to all clients (Arc zero-copy)
                for client in s.clients.read().await.iter() {
                    let _ = client.send(shared.clone()).await;
                }
                
                // Add to scrollback
                drop(s);
                session.write().await.scrollback.push(Line::from_bytes(&shared));
            }
            None => break,  // EOF - PTY terminated
        }
    }
    
    // Mark session terminated
    session.write().await.state = SessionState::Terminated;
}
