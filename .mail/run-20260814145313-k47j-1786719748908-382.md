## 2. Tokio Integration Pattern

**Use this pattern:**

```rust
// pty/mod.rs
use tokio::io::{AsyncReadExt, AsyncWriteExt};
use tokio::sync::mpsc;
use windows::Win32::System::Console::*;

pub struct PtyHandle {
    hpc: HPCON,
    process_handle: HANDLE,
    output_rx: mpsc::Receiver<Vec<u8>>,
    input_tx: mpsc::Sender<Vec<u8>>,
}

impl PtyHandle {
    /// Spawn ConPTY session (async-friendly)
    pub async fn spawn(
        shell: &str,
        working_dir: &Path,
        rows: u16,
        cols: u16,
    ) -> Result<Self, PtyError> {
        // Create pipes (FILE_FLAG_OVERLAPPED for async)
        let (input_read, input_write) = create_pipe_async()?;
        let (output_read, output_write) = create_pipe_async()?;
        
        // CreatePseudoConsole (blocking FFI, but fast <1ms)
        let coord = COORD { X: cols as i16, Y: rows as i16 };
        let mut hpc = HPCON::default();
        unsafe {
            CreatePseudoConsole(coord, input_read, output_write, 0, &mut hpc)?;
        }
        
        // Spawn process with ConPTY attached
        let process_handle = spawn_process_with_conpty(shell, working_dir, hpc)?;
        
        // Close write-end handles (process owns them now)
        unsafe {
            CloseHandle(input_read);
            CloseHandle(output_write);
        }
        
        // Spawn async I/O tasks
        let (output_tx, output_rx) = mpsc::channel(256);
        let (input_tx, mut input_rx) = mpsc::channel(256);
        
        // Output reader task (ConPTY → Session Manager)
        let output_read = AsyncHandle::from_raw_handle(output_read)?;
        tokio::spawn(async move {
            let mut reader = tokio::io::BufReader::new(output_read);
            let mut buffer = vec![0u8; 4096];
            
            loop {
                match reader.read(&mut buffer).await {
                    Ok(0) => break,  // EOF
                    Ok(n) => {
                        let chunk = buffer[..n].to_vec();
                        if output_tx.send(chunk).await.is_err() {
                            break;  // Receiver dropped
                        }
                    }
                    Err(e) => {
                        tracing::error!("PTY read error: {}", e);
                        break;
                    }
                }
            }
        });
        
        // Input writer task (Session Manager → ConPTY)
        let input_write = AsyncHandle::from_raw_handle(input_write)?;
        tokio::spawn(async move {
            let mut writer = tokio::io::BufWriter::new(input_write);
            
            while let Some(data) = input_rx.recv().await {
                if let Err(e) = writer.write_all(&data).await {
                    tracing::error!("PTY write error: {}", e);
                    break;
                }
                let _ = writer.flush().await;
            }
        });
        
        Ok(Self {
            hpc,
            process_handle,
            output_rx,
            input_tx,
        })
    }
    
    /// Read output (non-blocking)
    pub async fn read(&mut self) -> Option<Vec<u8>> {
        self.output_rx.recv().await
    }
    
    /// Write input
    pub async fn write(&self, data: &[u8]) -> Result<(), PtyError> {
        self.input_tx.send(data.to_vec()).await
            .map_err(|_| PtyError::Disconnected)
    }
    
    /// Resize terminal
    pub fn resize(&self, rows: u16, cols: u16) -> Result<(), PtyError> {
        let coord = COORD { X: cols as i16, Y: rows as i16 };
        unsafe {
            ResizePseudoConsole(self.hpc, coord)?;
        }
        Ok(())
    }
    
    /// Kill process
    pub fn kill(&self) -> Result<(), PtyError> {
        unsafe {
            TerminateProcess(self.process_handle, 1)?;
        }
        Ok(())
    }
}

impl Drop for PtyHandle {
    fn drop(&mut self) {
        unsafe {
            let _ = ClosePseudoConsole(self.hpc);
            let _ = CloseHandle(self.process_handle);
        }
    }
}

/// Helper: Create async pipe (FILE_FLAG_OVERLAPPED)
fn create_pipe_async() -> Result<(HANDLE, HANDLE), PtyError> {
    // Use CreatePipe with FILE_FLAG_OVERLAPPED
    // OR use named pipe with async flags
    // See Alacritty implementation for reference
    unimplemented!("Async pipe creation")
}
```

**Key points:**
- Use `tokio::spawn` for I/O tasks (not manual polling)
- mpsc channels bridge FFI ↔ async world
- 4KB buffer size per SRS §3.1.4
- Error handling: break on EOF, log and break on errors

## 3. Error Handling Strategy

**Define `PtyError` enum:**

```rust
// pty/error.rs
use thiserror::Error;

#[derive(Error, Debug)]
pub enum PtyError {
    #[error("ConPTY creation failed: {0}")]
    CreateFailed(#[from] windows::core::Error),
    
    #[error("Process spawn failed: {0}")]
    SpawnFailed(String),
    
    #[error("Resize failed: {0}")]
    ResizeFailed(windows::core::Error),
    
    #[error("PTY I/O error: {0}")]
    IoError(#[from] std::io::Error),
    
    #[error("PTY disconnected")]
    Disconnected,
    
    #[error("Invalid dimensions: rows={0}, cols={1}")]
    InvalidDimensions(u16, u16),
}

pub type Result<T> = std::result::Result<T, PtyError>;
```

**Non-fatal error handling:**
- `ResizePseudoConsole` fails → log warning, continue session (terminal might not resize but shell still works)
- `write()` fails → return error to caller (Session Manager decides whether to kill session)
- `read()` EOF → clean shutdown, notify Session Manager

**Fatal errors:**
- `CreatePseudoConsole` fails → propagate error, can't create session
- `TerminateProcess` fails → log error but continue cleanup
