```protobuf
syntax = "proto3";
package monoterminal.v1;

// ============================================================================
// Core Envelope (from SRS §3.1.1)
// ============================================================================

message Envelope {
  uint64 sequence_number = 1;
  oneof message {
    AttachRequest attach_request = 2;
    AttachResponse attach_response = 3;
    InputData input_data = 4;
    OutputData output_data = 5;
    ResizeRequest resize_request = 6;
    DetachRequest detach_request = 7;
    ErrorResponse error_response = 8;
  }
}

// ============================================================================
// Request/Response Messages
// ============================================================================

message AttachRequest {
  string session_id = 1;         // UUID or empty for new session
  string auth_token = 2;         // JWT
  uint32 rows = 3;               // Terminal dimensions
  uint32 cols = 4;
  uint64 last_seen_sequence = 5; // For late-joiner sync (0 = full scrollback)
}

message AttachResponse {
  string session_id = 1;
  SessionMetadata metadata = 2;
  repeated Line scrollback = 3;  // Last 10k lines (Phase 1: in-memory only)
}

message InputData {
  bytes data = 1;  // Raw keyboard input (UTF-8)
}

message OutputData {
  bytes data = 1;          // PTY output chunk
  uint64 sequence = 2;     // For ordering/dedup
  CompressionType compression = 3;
}

message ResizeRequest {
  uint32 rows = 1;
  uint32 cols = 2;
}

message DetachRequest {
  string session_id = 1;  // Session to detach from
}

message ErrorResponse {
  ErrorCode code = 1;
  string message = 2;
}

// ============================================================================
// Supporting Types
// ============================================================================

message SessionMetadata {
  string shell_type = 1;      // "cmd.exe", "powershell.exe", etc.
  string working_dir = 2;     // Current working directory
  uint32 rows = 3;            // Current dimensions
  uint32 cols = 4;
  int64 created_at = 5;       // Unix timestamp (seconds)
  int64 last_activity = 6;    // Unix timestamp (seconds)
}

message Line {
  bytes data = 1;             // Line content (UTF-8, with ANSI codes)
  uint64 line_number = 2;     // Sequential line number
}

// ============================================================================
// Enums
// ============================================================================

enum CompressionType {
  NONE = 0;
  ZSTD = 1;
}

enum ErrorCode {
  UNKNOWN = 0;
  SESSION_NOT_FOUND = 1;
  AUTH_FAILED = 2;
  PERMISSION_DENIED = 3;
  RATE_LIMIT_EXCEEDED = 4;
  INVALID_REQUEST = 5;
  SERVER_ERROR = 6;
}
```
