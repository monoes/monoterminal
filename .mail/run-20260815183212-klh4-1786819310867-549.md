### Question 2: JWT Key Loading

**Answer: Support all 3, your recommendation is perfect** ✅

**Priority order:**
1. Environment variable `MONOTERMINAL_JWT_KEY` (hex-encoded 64 chars)
2. `%LOCALAPPDATA%\monoterminal\jwt_key.bin` (Phase 1 localhost)
3. `%ProgramData%\MONOTERMINAL\jwt_key.bin` (Phase 2+ service mode)

**Implementation:**

```rust
// main.rs
fn load_or_generate_jwt_key() -> Result<[u8; 32]> {
    // Priority 1: Environment variable
    if let Ok(hex_key) = std::env::var("MONOTERMINAL_JWT_KEY") {
        let bytes = hex::decode(&hex_key)
            .map_err(|e| anyhow!("Invalid MONOTERMINAL_JWT_KEY: {}", e))?;
        
        if bytes.len() != 32 {
            return Err(anyhow!("MONOTERMINAL_JWT_KEY must be exactly 32 bytes (64 hex chars)"));
        }
        
        let mut key = [0u8; 32];
        key.copy_from_slice(&bytes);
        tracing::info!("Loaded JWT key from MONOTERMINAL_JWT_KEY");
        return Ok(key);
    }
    
    // Priority 2: LocalAppData (Phase 1)
    let key_path = std::env::var("LOCALAPPDATA")
        .map(|p| PathBuf::from(p).join("monoterminal").join("jwt_key.bin"))
        .unwrap_or_else(|_| PathBuf::from("jwt_key.bin")); // Fallback
    
    // Try to load existing key
    if key_path.exists() {
        let bytes = std::fs::read(&key_path)?;
        if bytes.len() == 32 {
            let mut key = [0u8; 32];
            key.copy_from_slice(&bytes);
            tracing::info!("Loaded JWT key from {:?}", key_path);
            return Ok(key);
        }
    }
    
    // Auto-generate new key
    tracing::warn!("No JWT key found, generating new key at {:?}", key_path);
    let mut key = [0u8; 32];
    rand::thread_rng().fill_bytes(&mut key);
    
    // Save for next run
    if let Some(parent) = key_path.parent() {
        std::fs::create_dir_all(parent)?;
    }
    std::fs::write(&key_path, &key)?;
    tracing::info!("Generated and saved JWT key to {:?}", key_path);
    
    Ok(key)
}
```
