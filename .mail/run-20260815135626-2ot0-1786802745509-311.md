```rust
// config.rs or auth/config.rs
fn load_jwt_signing_key() -> Result<[u8; 32]> {
    // 1. Try environment variable first (for testing/CI)
    if let Ok(key_hex) = env::var("MONOTERMINAL_JWT_KEY") {
        let key_bytes = hex::decode(key_hex)?;
        if key_bytes.len() == 32 {
            return Ok(key_bytes.try_into().unwrap());
        }
    }
    
    // 2. Try config file path
    let config = load_config()?;
    let key_path = shellexpand::tilde(&config.auth.jwt_signing_key_path);
    
    if Path::new(key_path.as_ref()).exists() {
        let key_bytes = fs::read(key_path.as_ref())?;
        if key_bytes.len() == 32 {
            return Ok(key_bytes.try_into().unwrap());
        }
    }
    
    // 3. Generate new key on first run
    warn!("No JWT signing key found, generating new key at {}", key_path);
    let mut key = [0u8; 32];
    rand::thread_rng().fill_bytes(&mut key);
    
    // Ensure directory exists
    if let Some(parent) = Path::new(key_path.as_ref()).parent() {
        fs::create_dir_all(parent)?;
    }
    
    // Save for future runs
    fs::write(key_path.as_ref(), &key)?;
    
    info!("Generated and saved new JWT signing key");
    Ok(key)
}
```
