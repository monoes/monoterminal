**2. Rate Limiting Hooks**

Three integration points:

```rust
use crate::auth::RateLimiter;

// Global rate limiter instance
let rate_limiter = RateLimiter::new();

// Hook 1: On new WebSocket connection
async fn on_connection(peer_addr: SocketAddr, rate_limiter: &RateLimiter) -> Result<()> {
    rate_limiter.check_connection(&peer_addr)
        .map_err(|e| ServerError::RateLimitExceeded(e.to_string()))?;
    Ok(())
}

// Hook 2: On auth attempt (challenge-response flow)
async fn on_auth_attempt(peer_addr: SocketAddr, rate_limiter: &RateLimiter) -> Result<()> {
    rate_limiter.check_auth_attempt(&peer_addr)
        .map_err(|e| ServerError::Banned(e.to_string()))?;
    Ok(())
}

// Hook 3: On auth failure
fn on_auth_failure(peer_addr: SocketAddr, rate_limiter: &RateLimiter) {
    rate_limiter.record_auth_failure(&peer_addr);
}

// Hook 4: On session create
async fn on_session_create(user_id: &str, rate_limiter: &RateLimiter) -> Result<()> {
    rate_limiter.check_session_create(user_id)
        .map_err(|e| ServerError::RateLimitExceeded(e.to_string()))?;
    Ok(())
}
```
