## Recommended: Add These Types to Match Backend

```typescript
// Top-level aggregate (matches Rust DashboardData)
export interface DashboardData {
  org_status: OrgStatus;
  agents: AgentInfo[];
  runs: RunInfo[];
  memory_stats: MemoryStats;
  timestamp: number; // epoch millis
}

// Org runtime state (matches Rust OrgStatus)
export interface OrgStatus {
  running: boolean;
  name?: string;
  run_id?: string;
  active_agents: number;
  pending_tasks: number;
  status_message: string;
}

// Per-agent details (matches Rust AgentInfo)
export interface AgentInfo {
  id: string;
  agent_type: string;
  status: string;
  tasks_completed: number;
  uptime_secs: number;
}

// Run history entry (matches Rust RunInfo)
export interface RunInfo {
  id: string;
  org_name: string;
  started_at: string; // ISO timestamp
  ended_at?: string;
  outcome: string;
  tokens: number;
}

// Memory stats (matches Rust MemoryStats)
export interface MemoryStats {
  total_entries: number;
  kg_nodes: number;
  kg_edges: number;
  db_size_bytes: number;
}

// Helper to parse DashboardResponse
export function parseDashboardData(response: DashboardResponse): DashboardData | null {
  if (response.error !== ErrorCode.SUCCESS) {
    return null;
  }
  
  try {
    return JSON.parse(response.jsonData) as DashboardData;
  } catch (e) {
    console.error('Failed to parse dashboard data:', e);
    return null;
  }
}
```
