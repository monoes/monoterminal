#### Hook 2: Working Directory Change

**Option A:** Parse PTY output for `cd` commands (simple but imprecise)
**Option B:** Track shell prompt changes (more robust)
**Option C:** Platform-specific cwd polling (most reliable)

**I recommend Option C** for Phase 1:

```rust
impl Session {
    /// Get current working directory of PTY process
    fn get_current_cwd(&self) -> Result<PathBuf> {
        #[cfg(target_os = "windows")]
        {
            // Read cwd from ConPTY process handle
            // (requires storing process handle in Session struct)
            todo!("Windows: Get PTY cwd via process handle")
        }
        
        #[cfg(unix)]
        {
            use std::fs;
            let cwd_link = format!("/proc/{}/cwd", self.pty.pid());
            fs::read_link(cwd_link)
                .context("Failed to read PTY cwd from /proc")
        }
    }
    
    /// Poll current cwd and trigger detection on change
    async fn poll_cwd_changes(&mut self) -> Result<()> {
        let mut last_cwd = self.working_dir.clone();
        
        loop {
            tokio::time::sleep(Duration::from_secs(2)).await;
            
            match self.get_current_cwd() {
                Ok(current_cwd) if current_cwd != last_cwd => {
                    // Working directory changed
                    tracing::debug!(
                        old = %last_cwd.display(),
                        new = %current_cwd.display(),
                        "Working directory changed"
                    );
                    
                    last_cwd = current_cwd.clone();
                    
                    // Trigger detection
                    tokio::spawn({
                        let cwd = current_cwd.clone();
                        async move {
                            use monoterminal_monomind_bridge::detect_monomind;
                            let detection = detect_monomind(&cwd);
                            
                            if detection.suggest_install {
                                tracing::info!(
                                    path = %cwd.display(),
                                    "Monomind not detected after cwd change"
                                );
                            }
                        }
                    });
                }
                Ok(_) => {
                    // No change
                }
                Err(e) => {
                    tracing::warn!(error = %e, "Failed to get current cwd");
                }
            }
        }
    }
}
```
