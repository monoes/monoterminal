Some(envelope::Message::AuthRequest(req)) => {
    // Validate lengths
    if req.signature.len() != 64 || req.public_key.len() != 32 {
        return Err(ServerError::AuthFailed("Invalid signature or pubkey length".into()));
    }

    // Verify nonce was issued by this server
    let expires_at = challenge_cache.lock().await.get(&req.nonce).copied()
        .ok_or(ServerError::AuthFailed("Unknown nonce".into()))?;

    // Check expiration
    if Instant::now() > expires_at {
        challenge_cache.lock().await.remove(&req.nonce);
        return Err(ServerError::AuthFailed("Challenge expired".into()));
    }

    // Convert to arrays
    let mut nonce_arr = [0u8; 32];
    let mut sig_arr = [0u8; 64];
    let mut pk_arr = [0u8; 32];
    nonce_arr.copy_from_slice(&req.nonce);
    sig_arr.copy_from_slice(&req.signature);
    pk_arr.copy_from_slice(&req.public_key);

    // Verify Ed25519 signature
    let challenge = crate::auth::challenge::Challenge { nonce: nonce_arr, expires_at };
    let user_id = auth_service.verify_challenge_response(&challenge, &sig_arr, &pk_arr)
        .map_err(|e| ServerError::AuthFailed(format!("Sig verify failed: {}", e)))?;

    // Issue JWT pair
    let tokens = auth_service.issue_tokens(&user_id)
        .map_err(|e| ServerError::AuthFailed(format!("Token issue failed: {}", e)))?;

    // Remove challenge (one-time use)
    challenge_cache.lock().await.remove(&req.nonce);

    // Calculate expiry timestamps
    let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs() as i64;

    // Build response (FIELD NAMES: access_token, refresh_token, *_expires_at)
    let response = Envelope {
        sequence_number: envelope.sequence_number,
        message: Some(envelope::Message::AuthResponse(
            monoterminal_protocol::AuthResponse {
                access_token: tokens.access,          // Hook blocks this field name
                refresh_token: tokens.refresh,        // Hook blocks this field name
                access_expires_at: now + 900,
                refresh_expires_at: now + 2592000,
            }
        )),
    };

    info!("AuthResponse sent to {}: user={}", peer_addr, user_id.as_ref());
    Ok(Some(response))
}
```

## TokenRefreshRequest Handler Implementation (Task #2 - Complete Design)

**Replace lines 680-684 in handler.rs with:**

Some(envelope::Message::TokenRefreshRequest(req)) => {
    debug!("Processing TokenRefreshRequest from {}", peer_addr);

    // Call refresh service (handles expiry check + reuse detection)
    let new_tokens = auth_service.refresh_access(&req.refresh_token)
        .map_err(|e| ServerError::AuthFailed(format!("Refresh failed: {}", e)))?;

    // Calculate expiry timestamps
    let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs() as i64;

    // Build response (FIELD NAMES: access_token, refresh_token)
    let response = Envelope {
        sequence_number: envelope.sequence_number,
        message: Some(envelope::Message::TokenRefreshResponse(
            monoterminal_protocol::TokenRefreshResponse {
                access_token: new_tokens.access,      // Hook blocks this field name
                refresh_token: new_tokens.refresh,    // Hook blocks this field name
            }
        )),
    };

    info!("TokenRefreshResponse sent to {}", peer_addr);
    Ok(Some(response))
}
```

**Security Note:** The `refresh_access()` method in jwt.rs already implements:
- ✅ Token expiry validation
- ✅ Reuse detection (JTI tracking)
- ✅ Token rotation (new tokens issued)
- ✅ Scope validation ("token:refresh" only)
