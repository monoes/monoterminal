```typescript
/**
 * Request monomind health check
 */
async sendHealthCheckRequest(projectDir: string): Promise<HealthCheckResponse> {
  return new Promise((resolve, reject) => {
    const envelope = {
      sequenceNumber: ++this.sequenceNumber,
      healthCheckRequest: { projectDir }
    };
    
    // Set one-time handler for response
    const originalHandler = this.messageHandlers.onHealthCheckResponse;
    this.messageHandlers.onHealthCheckResponse = (response) => {
      this.messageHandlers.onHealthCheckResponse = originalHandler;
      resolve(response);
    };
    
    this.sendEnvelope(envelope);
    
    // Timeout after 10s
    setTimeout(() => reject(new Error('Health check timeout')), 10000);
  });
}
```

#### 7. Update handleMessage to Route New Messages

```typescript
private handleMessage(data: ArrayBuffer): void {
  try {
    const buffer = new Uint8Array(data);
    const envelope: any = EnvelopeType.decode(buffer);
    const obj = EnvelopeType.toObject(envelope, {
      longs: Number,
      bytes: Uint8Array,
      defaults: true,
    });

    // Existing handlers
    if (obj.attachResponse && this.messageHandlers.onAttachResponse) {
      this.messageHandlers.onAttachResponse(obj.attachResponse);
    } else if (obj.outputData && this.messageHandlers.onOutputData) {
      this.messageHandlers.onOutputData(obj.outputData);
    } else if (obj.errorResponse && this.messageHandlers.onErrorResponse) {
      this.messageHandlers.onErrorResponse(obj.errorResponse);
    }
    // ADD THESE:
    else if (obj.healthCheckResponse && this.messageHandlers.onHealthCheckResponse) {
      this.messageHandlers.onHealthCheckResponse(obj.healthCheckResponse);
    } else if (obj.upgradeResponse && this.messageHandlers.onUpgradeResponse) {
      this.messageHandlers.onUpgradeResponse(obj.upgradeResponse);
    } else if (obj.detectionResponse && this.messageHandlers.onDetectionResponse) {
      this.messageHandlers.onDetectionResponse(obj.detectionResponse);
    } else if (obj.monitoringData && this.messageHandlers.onMonitoringData) {
      this.messageHandlers.onMonitoringData(obj.monitoringData);
    }
  } catch (error) {
    console.error('Failed to decode message:', error);
  }
}
```
