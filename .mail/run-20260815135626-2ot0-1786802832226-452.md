**Deliverable:** `crates/master/benches/fps_rendering.rs`

```rust
//! FPS rendering benchmark
//! Validates SRS §7.1.1: 60 FPS master rendering

use criterion::{black_box, criterion_group, criterion_main, Criterion};
use monoterminal_master::rendering::Renderer;

fn bench_fps_scrolling(c: &mut Criterion) {
    let mut group = c.benchmark_group("fps-rendering");
    
    // Simulate scrolling workload (1000 lines)
    let workload: Vec<String> = (0..1000)
        .map(|i| format!("Line {}: Terminal output with ANSI codes \x1b[32m{}\x1b[0m", i, "█".repeat(80)))
        .collect();
    
    group.bench_function("render-scrolling-1000-lines", |b| {
        let mut renderer = Renderer::new(/* config */);
        
        b.iter(|| {
            for line in &workload {
                renderer.render_line(black_box(line));
            }
            renderer.present();  // Present frame
        });
    });
    
    group.finish();
}

criterion_group!(benches, bench_fps_scrolling);
criterion_main!(benches);
```

**Success Criteria:**
- ✅ p50 FPS ≥ 60
- ✅ p95 FPS ≥ 58
- ✅ Tested on Windows 10 1809 + Windows 11 23H2

### Task #12 (task-9): Latency Benchmark - Criterion #5

**Objective:** Validate <10ms LAN latency p95 (SRS §7.1.5)

**Deliverable:** `crates/master/benches/websocket_latency.rs`

```rust
//! WebSocket round-trip latency benchmark
//! Validates SRS §7.1.5: <10ms local latency (p95)

use criterion::{black_box, criterion_group, criterion_main, Criterion, BenchmarkId};
use tokio::runtime::Runtime;

fn bench_websocket_rtt(c: &mut Criterion) {
    let rt = Runtime::new().unwrap();
    let mut group = c.benchmark_group("websocket-latency");
    group.sample_size(10_000);  // Large sample for p95 accuracy
    
    group.bench_function("echo-rtt-p95", |b| {
        b.to_async(&rt).iter_custom(|iters| async move {
            let mut client = WebSocketClient::connect("ws://localhost:8080").await.unwrap();
            let start = Instant::now();
            
            for _ in 0..iters {
                client.send_input(b"X\n").await.unwrap();
                let _response = client.recv_output().await.unwrap();
            }
            
            start.elapsed()
        });
    });
    
    group.finish();
}

criterion_group!(benches, bench_websocket_rtt);
criterion_main!(benches);
```

**Success Criteria:**
- ✅ p50 < 5ms, p95 < 10ms, p99 < 15ms
- ✅ Wireshark packet capture validates (manual verification)
- ✅ Consistent under 10 concurrent clients

**Evidence:** Criterion HTML report + Wireshark PCAP file

### Task #13 (task-10): 24-Hour Soak Test - Criterion #7

**Objective:** Validate zero crashes in 24h stability test (SRS §7.1.7)

**Deliverable:** `tests/soak/24h-stability.rs`

```rust
//! 24-hour soak test for stability validation
//! Run with: cargo test --release --test soak --ignored -- --nocapture

use monoterminal_master::Master;
use std::time::{Duration, Instant};
use tokio::time::sleep;

#[tokio::test]
#[ignore]  // Only run manually or in CI on main branch
async fn test_24h_zero_crashes() {
    const DURATION: Duration = Duration::from_secs(86400);  // 24 hours
    const SESSION_CREATE_INTERVAL: Duration = Duration::from_secs(300);  // 5 minutes
    
    let master = Master::start(/* config */).await.unwrap();
    let start = Instant::now();
    let mut iteration = 0;
    
    while start.elapsed() < DURATION {
        iteration += 1;
        println!("[{}] Iteration {}/288", start.elapsed().as_secs(), iteration);
        
        // Create 10 sessions
        let mut sessions = vec![];
        for i in 0..10 {
            let session = master.create_session(&format!("soak-{}-{}", iteration, i)).await.unwrap();
            sessions.push(session);
        }
        
        // Send random input
        for session in &sessions {
            session.send_input(b"echo 'Soak test iteration'\n").await.unwrap();
        }
        
        // Terminate 5 sessions randomly
        for session in sessions.iter().take(5) {
            session.terminate().await.unwrap();
        }
        
        // Monitor memory
        let mem_mb = get_process_memory_mb();
        println!("  Memory: {} MB", mem_mb);
        
        sleep(SESSION_CREATE_INTERVAL).await;
    }
    
    // Assert master still running
    assert!(master.is_running(), "Master crashed during soak test!");
    println!("✅ 24-hour soak test PASSED - zero crashes");
}
```

**Monitoring Script:** `tests/soak/monitor-memory.ps1`
