## Vertex Buffer Dynamic Resize - Design

**Current Issue:** Fixed size buffer (1920 × 6 = 80×24 grid max)

**Proposed Fix (Tuesday session):**
```rust
pub fn resize(&mut self, width: u32, height: u32) -> Result<()> {
    // ... existing surface resize ...
    
    // Calculate new grid dimensions from window size
    let cols = (width / self.cell_width).max(1);
    let rows = (height / self.cell_height).max(1);
    
    // Resize terminal grid
    self.terminal_grid.resize(rows as u16, cols as u16);
    
    // Reallocate vertex buffer for new size
    let max_vertices = (rows * cols * 6) as u64;
    let vertex_buffer_size = max_vertices * std::mem::size_of::<GlyphVertex>() as u64;
    
    if let Some(device) = &self.device {
        self.vertex_buffer = Some(device.create_buffer(&wgpu::BufferDescriptor {
            label: Some("Terminal Vertex Buffer"),
            size: vertex_buffer_size,
            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
            mapped_at_creation: false,
        }));
        
        tracing::info!("Vertex buffer resized: {}×{} grid = {} vertices", 
            cols, rows, max_vertices);
    }
    
    Ok(())
}
```
