```rust
// crates/master/src/ui_bridge.rs (NEW FILE)
//
// Bridge layer between UI (wgpu/egui) and SessionManager
// Provides thread-safe access from main UI thread to tokio SessionManager

pub struct UiBridge {
    session_manager: Arc<SessionManager>,
    runtime: tokio::runtime::Runtime, // UI thread → tokio bridge
}

impl UiBridge {
    /// Create from existing SessionManager
    pub fn new(session_manager: Arc<SessionManager>) -> Self {
        Self {
            session_manager,
            runtime: tokio::runtime::Runtime::new().unwrap(),
        }
    }

    /// Get session list (blocks UI thread briefly - acceptable for 60 FPS)
    pub fn get_sessions(&self) -> Vec<SessionInfo> {
        self.runtime.block_on(self.session_manager.list_sessions())
    }

    /// Get render data for active session
    pub fn get_render_data(&self, session_id: SessionId) -> Result<SessionRenderData> {
        self.runtime.block_on(self.session_manager.get_session_render_data(session_id))
    }
}
```
