### **2. Tue AM: fontdue Rasterization Code**

I'll provide by EOD Monday:
```rust
// In fonts.rs
use fontdue::{Font, FontSettings};

pub struct FontManager {
    font: Font,
    size: f32,  // e.g., 14.0
}

impl FontManager {
    pub fn new_consolas(size: f32) -> Result<Self> {
        // Load Consolas from Windows system fonts
        let font_data = Self::load_consolas_from_system()?;
        let font = Font::from_bytes(font_data, FontSettings::default())?;
        Ok(Self { font, size })
    }

    /// Rasterize single glyph
    pub fn rasterize(&self, ch: char) -> RasterizedGlyph {
        let (metrics, bitmap) = self.font.rasterize(ch, self.size);
        RasterizedGlyph {
            width: metrics.width,
            height: metrics.height,
            bearing_x: metrics.xmin,
            bearing_y: metrics.ymin,
            advance: metrics.advance_width as u32,
            bitmap,  // Vec<u8> R8 grayscale
        }
    }
}
```

You integrate this with your texture upload.
